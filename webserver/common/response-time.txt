# Kibana Dev Tools — 웹서버 응답시간 수집 (nginx·NPM·apache 공통)
# 절차 설명: docs/response-time.md
# 순서: A(매핑) → B(@custom 파이프라인) → C(_simulate) → D(rollover) → 이후 Fleet 태그 추가, 서버 로그 포맷 변경
# 모두 Fleet이 제공하는 @custom 확장 지점만 사용한다. Integration 기본 파이프라인·템플릿은 수정하지 않는다.

############################################################
# A. 매핑 (@custom component template)
#    먼저 기존 내용을 확인한다. 이미 사용자 설정이 있으면 그 내용에 아래 properties를 합쳐서 PUT 한다.
############################################################
GET _component_template/logs-nginx.access@custom,logs-apache.access@custom

PUT _component_template/logs-nginx.access@custom
{
  "template": {
    "mappings": {
      "properties": {
        "event":     { "properties": { "duration": { "type": "long" } } },
        "webserver": { "properties": { "upstream_duration": { "type": "long" } } }
      }
    }
  }
}

PUT _component_template/logs-apache.access@custom
{
  "template": {
    "mappings": {
      "properties": {
        "event": { "properties": { "duration": { "type": "long" } } }
      }
    }
  }
}

############################################################
# B. @custom ingest pipeline
#    Integration 기본 파이프라인의 마지막 단계에서 자동으로 호출된다.
#    로그 줄 끝의 rt= / urt= 를 event.original 에서 읽는다.
#    접미어가 없는 줄(포맷 변경 전 로그)은 아무 처리 없이 통과한다.
############################################################
PUT _ingest/pipeline/logs-nginx.access@custom
{
  "description": "응답시간: rt=$request_time(초) → event.duration(ns), urt=\"$upstream_response_time\" → webserver.upstream_duration(ns)",
  "processors": [
    { "grok": {
        "field": "event.original",
        "patterns": [ " rt=%{NUMBER:_ws.rt} urt=\"%{DATA:_ws.urt}\"$" ],
        "ignore_missing": true, "ignore_failure": true } },
    { "script": {
        "if": "ctx._ws != null",
        "source": """
          if (ctx._ws.rt != null) {
            if (ctx.event == null) { ctx.event = [:]; }
            ctx.event.duration = Math.round(Double.parseDouble(ctx._ws.rt) * 1000000000L);
          }
          if (ctx._ws.urt != null) {
            // 여러 upstream을 거치면 "0.010, 0.020" 또는 "0.010 : 0.020" 형태 → 합산. "-"는 upstream 없음
            double total = 0; boolean found = false;
            for (def p : ctx._ws.urt.replace(':', ',').splitOnToken(',')) {
              String t = p.trim();
              if (t.length() > 0 && !t.equals('-')) { total += Double.parseDouble(t); found = true; }
            }
            if (found) {
              if (ctx.webserver == null) { ctx.webserver = [:]; }
              ctx.webserver.upstream_duration = Math.round(total * 1000000000L);
            }
          }
          ctx.remove('_ws');
        """,
        "ignore_failure": true } },
    { "remove": {
        "field": "event.original", "ignore_missing": true,
        "if": "ctx.tags instanceof List && ctx.tags.contains('preserve_original_event')" } },
    { "script": {
        "if": "ctx.tags instanceof List",
        "source": "ctx.tags.removeIf(t -> t == 'preserve_original_event')" } }
  ]
}

PUT _ingest/pipeline/logs-apache.access@custom
{
  "description": "응답시간: rt=%D(마이크로초) → event.duration(ns)",
  "processors": [
    { "grok": {
        "field": "event.original",
        "patterns": [ " rt=%{NUMBER:_ws.rt}$" ],
        "ignore_missing": true, "ignore_failure": true } },
    { "script": {
        "if": "ctx._ws != null",
        "source": """
          if (ctx._ws.rt != null) {
            if (ctx.event == null) { ctx.event = [:]; }
            ctx.event.duration = Long.parseLong(ctx._ws.rt) * 1000L;
          }
          ctx.remove('_ws');
        """,
        "ignore_failure": true } },
    { "remove": {
        "field": "event.original", "ignore_missing": true,
        "if": "ctx.tags instanceof List && ctx.tags.contains('preserve_original_event')" } },
    { "script": {
        "if": "ctx.tags instanceof List",
        "source": "ctx.tags.removeIf(t -> t == 'preserve_original_event')" } }
  ]
}

# D. 새 매핑을 적용하기 위한 rollover (A의 매핑은 다음 backing index부터 적용된다)

############################################################
# E. 수집 후 확인 — 도메인별 응답시간 (ms 환산)
############################################################
GET logs-nginx.access-*,logs-apache.access-*/_search
{
  "size": 0,
  "query": { "bool": { "filter": [
    { "range": { "@timestamp": { "gte": "now-1h" } } },
    { "exists": { "field": "event.duration" } } ] } },
  "aggs": { "ds": { "terms": { "field": "event.dataset" },
    "aggs": { "domain": { "terms": { "field": "destination.domain", "size": 10 },
      "aggs": {
        "p50_p95_p99_ns": { "percentiles": { "field": "event.duration", "percents": [50, 95, 99] } },
        "upstream_p95_ns": { "percentiles": { "field": "webserver.upstream_duration", "percents": [95] } }
      } } } } }
}

# E-1. 최신 문서에 event.original 과 preserve_original_event 태그가 남지 않았는지 (_source 직접 확인)
#      (event.original 은 색인하지 않는 필드라 exists 쿼리로는 확인할 수 없다)
GET logs-nginx.access-*,logs-apache.access-*/_search?filter_path=hits.hits._index,hits.hits._source
