# Kibana Dev Tools — apache Integration 파이프라인 파싱 검증 (P1, 설정 적용 전에 실행)
# Integration 설치 후 실행한다 (kibana.yml의 xpack.fleet.packages에 apache: latest 선언됨).

# 1) 설치된 파이프라인 이름 확인 (버전이 붙음, 예: logs-apache.access-<pkg-version>)
GET _ingest/pipeline/logs-apache.*?filter_path=*.description

# 2) access — 포맷 3종을 한 번에 시험 (<ACCESS_PIPELINE>을 1)에서 확인한 access 파이프라인 이름으로 바꾼다)
#    doc[0] ecs_vhost_combined  "%v:%p %a ..."   ← 1순위 (ecs-logging.conf 기본값)
#    doc[1] ecs_combined_noport "%v %a ..."      ← doc[0] 실패 시 2순위
#    doc[2] ecs_combined        "%a ..."         ← 기준선 (vhost 구분 없음)
#    doc[3] ecs_vhost_combined, IPv6 클라이언트, 응답 바이트 "-"

# 판정 기준
#   doc[0], doc[3]: destination.domain (+ destination.port), source.ip, http.request.method,
#                   http.response.status_code, url.original 이 있고 error 가 없으면 → ecs_vhost_combined 사용 (기본값 유지)
#   doc[0] 실패, doc[1] 성공 → ecs-logging.conf 의 GlobalLog 포맷을 ecs_combined_noport 로 변경
#   doc[0], doc[1] 모두 실패 → ecs_combined 사용 (vhost 구분은 destination.domain 대신 host.name/tags 로)
#   ※ doc[1] 형식은 "<도메인> <IP>" 가 "<IP1> <IP2>" 와 구분되지 않으므로, 도메인이 IP로 찍히는 경우가 없는지도 확인
